# Dependencias de execucao do ForensicGuard.
#
# As versoes sao fixadas (==) de proposito: reprodutibilidade.
# Um "fastapi>=0.141" instalaria uma versao diferente amanha e o projeto
# poderia quebrar sem que nenhuma linha de codigo tivesse mudado.
#
# Regra do projeto: uma biblioteca so entra aqui na etapa em que passa a ser
# usada de fato.

fastapi==0.141.1
uvicorn[standard]==0.52.4
pydantic==2.13.5
pydantic-settings==2.15.0

# Leitura de uploads multipart/form-data. O FastAPI nao a instala por padrao,
# mas exige sua presenca em qualquer rota que receba arquivo.
python-multipart==0.0.32

# Deteccao de tipo de arquivo em Python puro, sem bibliotecas nativas.
# Usada como segunda opiniao: a tabela propria em app/constants/signatures.py
# decide primeiro, e o puremagic responde pelos formatos que ela nao cobre.
# Substitui o python-magic, que exige libmagic/DLL e quebra no Windows.
puremagic==2.2.0

# Leitura de imagens: dimensoes, modo de cor, EXIF e GPS.
#
# O exifread, citado na especificacao, ficou de fora: o Pillow ja cobre tudo
# o que o MVP precisa, e uma dependencia a menos e uma superficie de ataque a
# menos num projeto que processa arquivos hostis.
Pillow==12.3.0

# Cliente HTTP, usado APENAS para resolver redirecionamentos quando a pessoa
# pede explicitamente. Nenhuma outra parte da ferramenta acessa a rede.
#
# Subiu de requirements-dev.txt para ca: ja era dependencia de teste, por ser
# o cliente que o TestClient do Starlette 1.6 exige, e agora passou a ser
# usado em execucao tambem. E a mesma biblioteca - o projeto nao mantem dois
# clientes HTTP para a mesma funcao.
httpx2==2.12.0
